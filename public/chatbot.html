📋 Replit Quick Setup
// server.js (Node + Express on Replit) const express = require('express'); const app = express(); app.use(express.json()); app.post('/chat', async (req, res) => { const resp = await fetch( 'https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'x-api-key': process.env.ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01', 'content-type': 'application/json', }, body: JSON.stringify({ model: 'claude-sonnet-4-20250514', max_tokens: 1024, messages: req.body.messages, system: req.body.system, }), } ); const data = await resp.json(); res.json(data); }); app.listen(3000);
